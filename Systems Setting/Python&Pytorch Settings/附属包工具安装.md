sudo apt install python3-colcon-common-extensions

# 使用帮助命令查看 colcon 信息
colcon --help

# 通过 apt 查看已安装的 colcon 版本
apt show python3-colcon-common-extensions | grep Version

# 或者使用 dpkg
dpkg -l | grep colcon

在虚拟环境里安装完pytorch后

安装附属工具：（按需要）

主环境：
# 系统级开发工具
pip install --user jupyterlab        # Jupyter实验室
pip install --user black            # 代码格式化
pip install --user flake8           # 代码检查
pip install --user pre-commit       # Git钩子
pip install --user tqdm             # 进度条
pip install --user rich             # 丰富终端输出

# 全局命令行工具
pip install --user tensorboard      # 训练可视化
pip install --user wandb            # 实验跟踪
pip install --user torchserve       # 模型服务

# 系统管理工具
pip install --user pipx             # 应用包管理
pip install --user virtualenv       # 虚拟环境管理

虚拟环境：
# 项目特定依赖
pip install torch torchvision torchaudio  # PyTorch核心
pip install numpy pandas scipy           # 科学计算
pip install matplotlib seaborn           # 绘图（项目特定版本）
pip install albumentations kornia        # 计算机视觉
pip install transformers datasets        # NLP工具

# 版本敏感的工具
pip install pytorch-lightning           # 训练框架
pip install timm efficientnet-pytorch   # 模型库
