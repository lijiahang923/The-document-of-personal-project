#!/bin/bash
# DRL-ROS2 导航管理器 - 最终稳定版
# 基于已验证的成功配置

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
VENV_PATH="$PROJECT_ROOT/drl_ros2_venv"
WORKSPACE_PATH="$PROJECT_ROOT/DRL-Robot-Navigation-ROS2"

echo "🤖 DRL-ROS2 导航管理器"
echo "========================"

# 关键修复：正确的 WSL2 图形配置
setup_graphics() {
    export DISPLAY=:0
    export LIBGL_ALWAYS_INDIRECT=0
    export GDK_BACKEND=x11
    export MESA_GL_VERSION_OVERRIDE=3.3
    export GZ_PHYSICS_ENGINE=ode
    
    echo "✅ 图形配置: DISPLAY=:0, LIBGL_ALWAYS_INDIRECT=0"
}

# 安全清理
safe_clean() {
    echo "🧹 清理环境..."
    pkill -f gzserver 2>/dev/null
    pkill -f gzclient 2>/dev/null
    pkill -f "ros2 launch" 2>/dev/null
    sleep 2
    echo "✅ 清理完成"
}

# ROS2 环境设置
setup_ros2() {
    source /opt/ros/humble/setup.bash
    if [ -f "$WORKSPACE_PATH/install/setup.bash" ]; then
        source "$WORKSPACE_PATH/install/setup.bash"
    fi
    
    export ROS_DOMAIN_ID=1
    export TURTLEBOT3_MODEL=waffle
    
    echo "✅ ROS2 环境配置完成"
}

# Python 环境设置
setup_python() {
    if [ -f "$VENV_PATH/bin/python3" ]; then
        export VIRTUAL_ENV="$VENV_PATH"
        export PATH="$VENV_PATH/bin:$PATH"
        echo "✅ Python 环境配置完成"
    else
        echo "❌ Python 虚拟环境不存在"
        return 1
    fi
}

# 图形功能测试
test_graphics() {
    echo "🧪 测试图形功能..."
    
    if xdpyinfo > /dev/null 2>&1; then
        echo "✅ X11 连接正常"
    else
        echo "❌ X11 连接失败 - 请检查 VcXsrv 配置"
        return 1
    fi
    
    echo "✅ 图形测试通过"
}

# 1. 分步启动训练世界（最稳定）
start_training_world_step() {
    echo "🔧 分步启动训练世界（最稳定）..."
    safe_clean
    setup_graphics
    setup_ros2
    
    echo "步骤 1: 启动 Gazebo 服务器（训练世界）..."
    # 使用训练世界的启动文件，但只启动服务器部分
    ros2 launch turtlebot3_gazebo ros2_drl.launch.py headless:=true &
    LAUNCH_PID=$!
    sleep 10
    
    if ! ps -p $LAUNCH_PID >/dev/null; then
        echo "❌ Gazebo 服务器启动失败"
        return 1
    fi
    
    echo "步骤 2: 启动 Gazebo 客户端..."
    gzclient --verbose &
    CLIENT_PID=$!
    
    echo ""
    echo "🎉 训练世界启动完成！"
    echo "📊 进程信息:"
    echo "   - ROS2 启动器: PID $LAUNCH_PID"
    echo "   - Gazebo 客户端: PID $CLIENT_PID"
    echo ""
    echo "💡 按 Ctrl+C 停止所有进程"
    
    # 等待用户中断
    wait
}

# 2. 直接启动训练世界
start_training_world() {
    echo "🚀 直接启动训练世界..."
    safe_clean
    setup_graphics
    test_graphics
    setup_ros2
    
    LAUNCH_FILE="$WORKSPACE_PATH/install/turtlebot3_gazebo/share/turtlebot3_gazebo/launch/ros2_drl.launch.py"
    if [ -f "$LAUNCH_FILE" ]; then
        echo "使用启动文件: $LAUNCH_FILE"
        ros2 launch "$LAUNCH_FILE"
    else
        echo "使用默认训练世界启动"
        ros2 launch turtlebot3_gazebo ros2_drl.launch.py
    fi
}

# 3. 基础 Gazebo 测试（空世界）
start_gazebo_basic() {
    echo "🌍 启动基础 Gazebo（空世界）..."
    safe_clean
    setup_graphics
    test_graphics
    
    gzserver --verbose &
    SERVER_PID=$!
    sleep 5
    
    if ps -p $SERVER_PID >/dev/null; then
        echo "✅ 服务器运行中，启动客户端..."
        gzclient --verbose
    else
        echo "❌ 服务器启动失败"
        return 1
    fi
}

# 4. DRL 训练启动
start_training() {
    echo "🎯 启动 DRL 训练..."
    setup_ros2
    setup_python
    
    cd "$WORKSPACE_PATH"
    python3 src/drl_navigation_ros2/train.py
}

# 5. TensorBoard 监测
start_tensorboard() {
    echo "📊 启动 TensorBoard 监测..."
    setup_python
    tensorboard --logdir "$WORKSPACE_PATH/runs" --host 0.0.0.0
}

# 6. RViz2 可视化（修复版）
start_rviz() {
    echo "👁️  启动 RViz2 可视化..."
    setup_ros2
    
    # 检查 turtlebot3_bringup 包是否存在
    if ros2 pkg list | grep -q "turtlebot3_bringup"; then
        echo "✅ 找到 turtlebot3_bringup 包，启动 RViz2..."
        ros2 launch turtlebot3_bringup rviz2.launch.py
    else
        echo "❌ turtlebot3_bringup 包未找到"
        echo "💡 正在安装缺失的包..."
        sudo apt update
        sudo apt install -y ros-humble-turtlebot3-bringup
        
        if [ $? -eq 0 ]; then
            echo "✅ 安装成功，重新启动 RViz2..."
            ros2 launch turtlebot3_bringup rviz2.launch.py
        else
            echo "❌ 安装失败，使用备用方案启动 RViz2..."
            echo "启动独立 RViz2 实例..."
            rviz2
        fi
    fi
}

# 7. 安装依赖
install_dependencies() {
    echo "📦 安装系统依赖..."
    sudo apt update
    sudo apt install -y \
        ros-humble-turtlebot3-bringup \
        ros-humble-navigation2 \
        ros-humble-nav2-bringup
    
    echo "✅ 依赖安装完成"
}

# 8. 环境诊断

    echo "🔍 环境诊断报告"
    echo "================"
    
    echo "1. 系统信息:"
    echo "   - 系统: $(lsb_release -d | cut -f2 2>/dev/null || echo '未知')"
    echo "   - 内核: $(uname -r)"
    echo "   - WSL2: 是"
    
    echo ""
    echo "2. 图形配置:"
    echo "   - DISPLAY: $DISPLAY"
    echo "   - LIBGL_ALWAYS_INDIRECT: ${LIBGL_ALWAYS_INDIRECT:-未设置}"
    echo "   - X11 测试: $(xdpyinfo >/dev/null 2>&1 && echo '✅ 正常' || echo '❌ 失败')"
    
    echo ""
    echo "3. 软件环境:"
    echo "   - ROS2: $(command -v ros2 >/dev/null && echo '✅ 已安装' || echo '❌ 未安装')"
    echo "   - Gazebo: $(command -v gzserver >/dev/null && echo '✅ 已安装' || echo '❌ 未安装')"
    echo "   - RViz2: $(command -v rviz2 >/dev/null && echo '✅ 已安装' || echo '❌ 未安装')"
    echo "   - Python虚拟环境: $([ -f "$VENV_PATH/bin/python3" ] && echo '✅ 存在' || echo '❌ 缺失')"
    
    echo ""
    echo "4. ROS2 包检查:"
    echo "   - turtlebot3_gazebo: $(ros2 pkg list | grep -q "turtlebot3_gazebo" && echo '✅ 存在' || echo '❌ 缺失')"
    echo "   - turtlebot3_bringup: $(ros2 pkg list | grep -q "turtlebot3_bringup" && echo '✅ 存在' || echo '❌ 缺失')"
    echo "   - 工作空间: $([ -f "$WORKSPACE_PATH/install/setup.bash" ] && echo '✅ 已构建' || echo '❌ 未构建')"
    
    echo ""
    echo "🎯 推荐启动方式:"
    echo "   - 最稳定: ./manage_project.sh training-world-step  (训练世界分步启动)"
    echo "   - 标准: ./manage_project.sh training-world        (训练世界直接启动)"
    echo "   - 测试: ./manage_project.sh gazebo                (空世界测试)"
    echo "   - 训练: ./manage_project.sh train"
}

# 9. 修复显示问题
fix_display_issues() {
    echo "🔧 修复显示配置..."
    
    # 更新 .bashrc 中的配置
    if ! grep -q "LIBGL_ALWAYS_INDIRECT=0" ~/.bashrc; then
        cat >> ~/.bashrc << 'EOL'

# DRL-ROS2 WSL2 图形配置
export DISPLAY=:0
export LIBGL_ALWAYS_INDIRECT=0
export GDK_BACKEND=x11
EOL
        echo "✅ 已更新 .bashrc 配置"
    else
        echo "✅ 配置已存在"
    fi
    
    source ~/.bashrc
    
    echo ""
    echo "💡 VcXsrv 配置建议:"
    echo "   - 显示编号: 0"
    echo "   - 勾选: Disable access control"
    echo "   - 勾选: Native opengl"
    echo "   - 取消: Disable composition"
}
