cat > manage_project.sh << 'EOF'
#!/bin/bash
# DRL-ROS2 项目管理脚本

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
VENV_PATH="$PROJECT_ROOT/drl_nav_env"
SCRIPTS_DIR="$PROJECT_ROOT/scripts"

echo "🤖 DRL-ROS2 导航项目管理器"
echo "================================"

case "${1:-}" in
    "activate")
        echo "激活虚拟环境..."
        source "$VENV_PATH/bin/activate"
        echo "虚拟环境已激活"
        ;;
    "deactivate")
        echo "停用虚拟环境..."
        deactivate
        ;;
    "train")
        echo "启动训练..."
        source "$VENV_PATH/bin/activate"
        cd "$HOME/DRL-Robot-Navigation-ROS2"
        python src/drl_navigation_ros2/train.py
        ;;
    "gazebo")
        echo "启动Gazebo..."
        source /opt/ros/humble/setup.bash
        export ROS_DOMAIN_ID=1
        export TURTLEBOT3_MODEL=waffle
        ros2 launch turtlebot3_gazebo ros2_drl.launch.py
        ;;
    "tensorboard")
        echo "启动TensorBoard..."
        source "$VENV_PATH/bin/activate"
        cd "$HOME/DRL-Robot-Navigation-ROS2"
        tensorboard --logdir runs --host 0.0.0.0
        ;;
    "check")
        echo "检查环境..."
        source "$VENV_PATH/bin/activate"
        python "$PROJECT_ROOT/scripts/check_environment.py"
        ;;
    "backup")
        echo "备份环境..."
        source "$VENV_PATH/bin/activate"
        "$PROJECT_ROOT/scripts/backup_environment.sh"
        ;;
    "install-deps")
        echo "安装项目依赖..."
        source "$VENV_PATH/bin/activate"
        pip install -r "$PROJECT_ROOT/requirements.txt"
        ;;
    *)
        echo "可用命令:"
        echo "  activate      - 激活虚拟环境"
        echo "  deactivate    - 停用虚拟环境" 
        echo "  train         - 启动训练"
        echo "  gazebo        - 启动Gazebo仿真"
        echo "  tensorboard   - 启动TensorBoard"
        echo "  check         - 检查环境完整性"
        echo "  backup        - 备份环境配置"
        echo "  install-deps  - 安装依赖"
        ;;
esac
EOF

chmod +x manage_project.sh

cat > ~/projects/drl_ros2_navigation/scripts/check_environment.py << 'EOF'
#!/usr/bin/env python3
"""
DRL-ROS2 环境完整性检查
"""

import sys
import subprocess

def run_command(cmd):
    """运行命令并返回输出"""
    try:
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
        return result.returncode == 0, result.stdout, result.stderr
    except Exception as e:
        return False, "", str(e)

def check_python_environment():
    """检查Python环境"""
    print("🐍 Python环境检查")
    print("-" * 40)
    
    # 检查Python版本
    success, stdout, stderr = run_command("python --version")
    if success:
        print(f"✅ Python版本: {stdout.strip()}")
    else:
        print(f"❌ Python检查失败: {stderr}")
    
    # 检查虚拟环境
    success, stdout, stderr = run_command("which python")
    if "drl_nav_env" in stdout:
        print("✅ 在项目虚拟环境中")
    else:
        print("⚠️  不在项目虚拟环境中")

def check_packages():
    """检查关键包"""
    print("\n📦 关键包检查")
    print("-" * 40)
    
    packages = [
        ("torch", "PyTorch"),
        ("torchvision", "TorchVision"),
        ("numpy", "NumPy"),
        ("cv2", "OpenCV"),
        ("tensorboard", "TensorBoard"),
        ("tqdm", "进度条"),
        ("albumentations", "图像增强"),
    ]
    
    all_ok = True
    for pkg, name in packages:
        try:
            __import__(pkg)
            version = getattr(__import__(pkg), '__version__', '未知版本')
            print(f"✅ {name:15} - {version}")
        except ImportError:
            print(f"❌ {name:15} - 缺失")
            all_ok = False
    
    return all_ok

def check_ros2():
    """检查ROS2环境"""
    print("\n🤖 ROS2环境检查")
    print("-" * 40)
    
    # 检查ROS2安装
    success, stdout, stderr = run_command("ros2 --version")
    if success:
        print(f"✅ ROS2: {stdout.strip()}")
    else:
        print("❌ ROS2未安装或未配置")
        return False
    
    # 检查必要的ROS2包
    ros_packages = [
        "turtlebot3_gazebo",
        "drl_navigation_ros2"
    ]
    
    for pkg in ros_packages:
        success, stdout, stderr = run_command(f"ros2 pkg list | grep {pkg}")
        if success and pkg in stdout:
            print(f"✅ ROS包: {pkg}")
        else:
            print(f"❌ ROS包缺失: {pkg}")
    
    return True

if __name__ == "__main__":
    print("🔍 DRL-ROS2 环境完整性检查")
    print("=" * 50)
    
    check_python_environment()
    packages_ok = check_packages()
    ros_ok = check_ros2()
    
    print("\n" + "=" * 50)
    if packages_ok and ros_ok:
        print("🎉 所有检查通过！环境准备就绪。")
    else:
        print("⚠️  请检查并修复上述问题。")
        sys.exit(1)
EOF



cat > ~/projects/drl_ros2_navigation/scripts/backup_environment.sh << 'EOF'
#!/bin/bash
# 环境备份脚本

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
BACKUP_DIR="$PROJECT_ROOT/backups"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/environment_backup_$DATE"

echo "📦 开始备份环境..."
mkdir -p "$BACKUP_DIR"

# 备份包列表
pip freeze > "${BACKUP_FILE}_requirements.txt"

# 备份环境信息
{
    echo "备份时间: $(date)"
    echo "项目路径: $PROJECT_ROOT"
    echo "Python版本: $(python --version 2>&1)"
    echo "虚拟环境: $(which python)"
} > "${BACKUP_FILE}_info.txt"

# 备份关键配置文件
if [ -f "$HOME/DRL-Robot-Navigation-ROS2/package.xml" ]; then
    cp "$HOME/DRL-Robot-Navigation-ROS2/package.xml" "${BACKUP_FILE}_package.xml"
fi

echo "✅ 环境已备份:"
echo "   - 包列表: ${BACKUP_FILE}_requirements.txt"
echo "   - 环境信息: ${BACKUP_FILE}_info.txt"
echo "   - 包含 $(wc -l < "${BACKUP_FILE}_requirements.txt") 个包"
EOF

chmod +x ~/projects/drl_ros2_navigation/scripts/backup_environment.sh
