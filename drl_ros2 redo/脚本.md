#!/bin/bash
# DRL-ROS2 项目管理脚本 - 完整版

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
VENV_PATH="$PROJECT_ROOT/drl_ros2_venv"
WORKSPACE_PATH="$PROJECT_ROOT/DRL-Robot-Navigation-ROS2"

echo "🤖 DRL-ROS2 导航项目管理器"
echo "================================"

case "${1:-}" in
    "activate")
        echo "激活虚拟环境..."
        source "$VENV_PATH/bin/activate"
        echo "虚拟环境已激活"
        ;;
    "deactivate")
        echo "停用虚拟环境..."
        deactivate
        ;;
    "build")
        echo "构建ROS2工作空间..."
        source "$VENV_PATH/bin/activate"
        cd "$WORKSPACE_PATH"
        colcon build --allow-overriding turtlebot3_description turtlebot3_gazebo
        echo "✅ 构建完成"
        ;;
    "train")
        echo "启动训练..."
        source "$VENV_PATH/bin/activate"
        source /opt/ros/humble/setup.bash
        source "$WORKSPACE_PATH/install/setup.bash"
        cd "$WORKSPACE_PATH"
        python src/drl_navigation_ros2/train.py
        ;;
    "gazebo")
        echo "启动Gazebo..."
        source /opt/ros/humble/setup.bash
        source "$WORKSPACE_PATH/install/setup.bash"
        export ROS_DOMAIN_ID=1
        export TURTLEBOT3_MODEL=waffle
        export GAZEBO_MODEL_PATH=$GAZEBO_MODEL_PATH:$WORKSPACE_PATH/src/turtlebot3_simulations/turtlebot3_gazebo/models
        ros2 launch turtlebot3_gazebo ros2_drl.launch.py
        ;;
    "tensorboard")
        echo "启动TensorBoard..."
        source "$VENV_PATH/bin/activate"
        cd "$WORKSPACE_PATH"
        tensorboard --logdir runs --host 0.0.0.0
        ;;
    "check")
        echo "检查环境..."
        source "$VENV_PATH/bin/activate"
        python "$PROJECT_ROOT/scripts/check_environment_robust.py"
        ;;
    "status")
        echo "项目状态..."
        echo "项目根目录: $PROJECT_ROOT"
        echo "虚拟环境: $VENV_PATH"
        if [ -d "$VENV_PATH" ]; then
            echo "✅ 虚拟环境存在"
        else
            echo "❌ 虚拟环境缺失"
        fi
        
        echo "ROS工作空间: $WORKSPACE_PATH"
        if [ -d "$WORKSPACE_PATH/install" ]; then
            echo "✅ ROS工作空间已构建"
        else
            echo "⚠️  ROS工作空间未构建"
        fi
        
        echo ""
        echo "📁 项目结构:"
        ls -la | grep -E "(drl_ros2_venv|scripts|logs|configs|backups|DRL-Robot-Navigation-ROS2)"
        ;;
    *)
        echo "可用命令:"
        echo "  activate    - 激活虚拟环境"
        echo "  deactivate  - 停用虚拟环境" 
        echo "  build       - 构建ROS2工作空间"
        echo "  train       - 启动训练"
        echo "  gazebo      - 启动Gazebo仿真"
        echo "  tensorboard - 启动TensorBoard"
        echo "  check       - 检查环境完整性"
        echo "  status      - 显示项目状态"
        ;;
esac

#!/usr/bin/env python3
"""
DRL-ROS2 环境完整性检查 (健壮版)
"""

import sys
import subprocess
import os

def run_command(cmd):
    """运行命令并返回输出"""
    try:
        full_cmd = f"source /opt/ros/humble/setup.bash && {cmd}"
        result = subprocess.run(
            full_cmd, 
            shell=True, 
            executable="/bin/bash",
            capture_output=True, 
            text=True
        )
        return result.returncode == 0, result.stdout, result.stderr
    except Exception as e:
        return False, "", str(e)

def check_ros2_installation():
    """检查ROS2安装"""
    print("🤖 ROS2安装检查")
    print("-" * 40)
    
    # 检查ROS2二进制文件
    ros2_path = "/opt/ros/humble/bin/ros2"
    if os.path.exists(ros2_path):
        print(f"✅ ROS2二进制文件存在: {ros2_path}")
        
        # 尝试获取版本 - 使用正确的命令
        success, stdout, stderr = run_command("ros2 version")
        if success:
            print(f"✅ ROS2版本: {stdout.strip().split()[0]}")
            return True
        else:
            print(f"⚠️  ROS2命令执行问题: {stderr}")
            # 即使命令有问题，只要二进制存在就继续
            return True
    else:
        print(f"❌ ROS2未安装在预期位置: {ros2_path}")
        return False

def check_python_environment():
    """检查Python环境"""
    print("\n🐍 Python环境检查")
    print("-" * 40)
    
    success, stdout, stderr = run_command("python --version")
    if success:
        print(f"✅ Python版本: {stdout.strip()}")
    
    success, stdout, stderr = run_command("which python")
    python_path = stdout.strip()
    print(f"Python路径: {python_path}")
    
    if "drl_ros2_venv" in python_path:
        print("✅ 在项目虚拟环境中")
    else:
        print("⚠️  不在项目虚拟环境中")
    
    return True

def check_packages():
    """检查关键包"""
    print("\n📦 关键包检查")
    print("-" * 40)
    
    packages = [
        ("torch", "PyTorch"),
        ("torchvision", "TorchVision"), 
        ("numpy", "NumPy"),
        ("cv2", "OpenCV"),
        ("tensorboard", "TensorBoard"),
        ("tqdm", "进度条"),
        ("albumentations", "图像增强"),
    ]
    
    all_ok = True
    for pkg, name in packages:
        try:
            module = __import__(pkg)
            version = getattr(module, '__version__', '未知版本')
            print(f"✅ {name:15} - {version}")
        except ImportError:
            print(f"❌ {name:15} - 缺失")
            all_ok = False
    
    return all_ok

def check_ros2_packages():
    """检查ROS2包"""
    print("\n📦 ROS2包检查")
    print("-" * 40)
    
    project_root = os.path.expanduser("~/projects/drl_ros2_navigation")
    workspace_setup = f"{project_root}/DRL-Robot-Navigation-ROS2/install/setup.bash"
    
    if os.path.exists(workspace_setup):
        cmd = f"source {workspace_setup} && ros2 pkg list"
        success, stdout, stderr = run_command(cmd)
        
        if success:
            if "drl_navigation_ros2" in stdout:
                print("✅ ROS包: drl_navigation_ros2")
            else:
                print("❌ ROS包缺失: drl_navigation_ros2")
                return False
        else:
            print(f"❌ 无法检查ROS包: {stderr}")
            return False
    else:
        print("❌ ROS工作空间未构建")
        return False
    
    # 检查其他必要包
    required_packages = ["turtlebot3_gazebo", "gazebo_ros"]
    all_packages_ok = True
    
    for pkg in required_packages:
        cmd = f"source {workspace_setup} && ros2 pkg list | grep {pkg}"
        success, stdout, stderr = run_command(cmd)
        if success and pkg in stdout:
            print(f"✅ ROS包: {pkg}")
        else:
            print(f"❌ ROS包缺失: {pkg}")
            all_packages_ok = False
    
    return all_packages_ok

if __name__ == "__main__":
    print("🔍 DRL-ROS2 环境完整性检查 (健壮版)")
    print("=" * 50)
    
    ros2_installed = check_ros2_installation()
    python_ok = check_python_environment()
    packages_ok = check_packages()
    
    if ros2_installed:
        ros_packages_ok = check_ros2_packages()
    else:
        ros_packages_ok = False
        print("\n⚠️  跳过ROS2包检查（ROS2未正确安装）")
    
    print("\n" + "=" * 50)
    if ros2_installed and python_ok and packages_ok and ros_packages_ok:
        print("🎉 所有检查通过！环境准备就绪。")
    else:
        print("❌ 环境检查未通过")
        if not ros2_installed:
            print("   - 请检查ROS2 Humble安装")
        if not ros_packages_ok:
            print("   - 请检查ROS2包构建")
        sys.exit(1)

#!/bin/bash
# 环境备份脚本

PROJECT_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
BACKUP_DIR="$PROJECT_ROOT/backups"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/environment_backup_$DATE"

echo "📦 开始备份环境..."
mkdir -p "$BACKUP_DIR"

# 备份包列表
pip freeze > "${BACKUP_FILE}_requirements.txt"

# 备份环境信息
{
    echo "备份时间: $(date)"
    echo "项目路径: $PROJECT_ROOT"
    echo "Python版本: $(python --version 2>&1)"
    echo "虚拟环境: $(which python)"
} > "${BACKUP_FILE}_info.txt"

# 备份关键配置文件
if [ -f "$HOME/DRL-Robot-Navigation-ROS2/package.xml" ]; then
    cp "$HOME/DRL-Robot-Navigation-ROS2/package.xml" "${BACKUP_FILE}_package.xml"
fi

echo "✅ 环境已备份:"
echo "   - 包列表: ${BACKUP_FILE}_requirements.txt"
echo "   - 环境信息: ${BACKUP_FILE}_info.txt"
echo "   - 包含 $(wc -l < "${BACKUP_FILE}_requirements.txt") 个包"
